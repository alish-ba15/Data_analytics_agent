## 5. Python Analysis Code & Executed Data Output  

```python
# -------------------------------------------------
# 1️⃣ Data cleaning (already applied, kept for context)
# -------------------------------------------------
clean_df = df.copy()
clean_df = clean_df.drop_duplicates()

# Strip whitespace
for col in clean_df.select_dtypes(include='object').columns:
    clean_df[col] = clean_df[col].astype(str).str.strip()

# Convert dates
clean_df['Transaction Date'] = pd.to_datetime(clean_df['Transaction Date'], errors='coerce')

# Normalize Discount Applied
clean_df['Discount Applied'] = clean_df['Discount Applied'].fillna(False).astype(bool)

# Recover missing numeric values
mask_total_missing = clean_df['Total Spent'].isna() & clean_df['Price Per Unit'].notna() & clean_df['Quantity'].notna()
clean_df.loc[mask_total_missing, 'Total Spent'] = (
    clean_df.loc[mask_total_missing, 'Price Per Unit'] *
    clean_df.loc[mask_total_missing, 'Quantity']
)

mask_price_missing = clean_df['Price Per Unit'].isna() & clean_df['Total Spent'].notna() & clean_df['Quantity'].notna()
clean_df.loc[mask_price_missing, 'Price Per Unit'] = (
    clean_df.loc[mask_price_missing, 'Total Spent'] /
    clean_df.loc[mask_price_missing, 'Quantity']
)

mask_qty_missing = clean_df['Quantity'].isna() & clean_df['Total Spent'].notna() & clean_df['Price Per Unit'].notna()
clean_df.loc[mask_qty_missing, 'Quantity'] = (
    clean_df.loc[mask_qty_missing, 'Total Spent'] /
    clean_df.loc[mask_qty_missing, 'Price Per Unit']
)

# Median imputation for any remaining NaNs
for col in ['Price Per Unit', 'Quantity', 'Total Spent']:
    clean_df[col] = clean_df[col].fillna(clean_df[col].median())

# Placeholder for missing Item names
clean_df['Item'] = clean_df['Item'].fillna('Unknown')

# -------------------------------------------------
# 2️⃣ Top‑5 categories by total revenue
# -------------------------------------------------
category_revenue = (
    clean_df.groupby('Category', as_index=False)['Total Spent']
            .sum()
            .rename(columns={'Total Spent': 'Total Revenue'})
)

result = category_revenue.sort_values('Total Revenue', ascending=False).head(5)
```
